class Dragnet::Value
  abstract do
    def text_value = :required
    def number_value = :optional
    def to_h = :optional
    def to_a = :optional
  end

  def ==(other)
    return false unless other.is_a?(self.class)
    
    text_value == other.text_value
  end

  def hash
    text_value.hash
  end

  def eql?(other)
    return false unless other.is_a?(self.class)

    super
  end

  def ===(other)
    text_value == other
  end
end
